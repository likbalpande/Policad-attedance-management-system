webhook_history
    id INT AUTO_INCREMENT PRIMARY KEY,
    attendance_history_id INT NOT NULL,
    status VARCHAR(50) NOT NULL, // application_level_enum(pending, completed)
    processed_at TIMESTAMPTZ,
    last_attempted_at TIMESTAMPTZ,
    processing_started_at TIMESTAMPTZ,
    is_processing BOOLEAN DEFAULT FALSE NOT NULL,
    retries INT DEFAULT 0,
    webhook_url TEXT NOT NULL,

    CONSTRAINT fk__webhook_history__attendance_history_id
        FOREIGN KEY (attendance_history_id) REFERENCES attendance_history(id),

    CREATE INDEX idx__webhook_history__status ON webhook_history(status) WHERE status != 'completed',
    CREATE INDEX idx__webhook_history__updated_at ON webhook_history(updated_at) WHERE status != 'completed',
    CREATE INDEX idx__webhook_history__processing_started_at ON webhook_history using BRIN (processing_started_at) WHERE status != 'completed',
    CREATE INDEX idx__webhook_history__retries ON webhook_history(retries) WHERE status != 'completed'
